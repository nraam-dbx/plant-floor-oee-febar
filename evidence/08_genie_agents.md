    AND `performance` IS NOT NULL
    AND `quality` IS NOT NULL
  GROUP BY `work_center`
), `ranked` AS (
  SELECT
    *,
    RANK() OVER (ORDER BY `avg_oee` ASC) AS `oee_rank`
  FROM `wc_metrics`

  `work_center`,
  `avg_oee`,
  `avg_availability`,
  `avg_performance`,
  `avg_quality`,

  END AS `biggest_loss_factor`,

  `avg_setup_min`,
  `avg_minor_stop_count`,
  `avg_scrap_qty`
FROM `ranked`
WHERE `oee_rank` <= 1
