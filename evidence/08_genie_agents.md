    AVG(`quality`) AS `avg_quality`

  WHERE `work_center` IS NOT NULL
    AND `oee` IS NOT NULL
    AND `availability` IS NOT NULL
    AND `performance` IS NOT NULL
    AND `quality` IS NOT NULL
  GROUP BY `work_center`
), ranked AS (
  SELECT
    `work_center`,
    `avg_oee`,
    `avg_availability`,
    `avg_performance`,
    `avg_quality`,
    RANK() OVER (ORDER BY `avg_oee` ASC) AS `oee_rank`
  FROM work_center_avgs
)
SELECT
  `work_center`,
  `avg_oee` * 100 AS `avg_oee_pct`,
  CASE
    WHEN `avg_availability` <= `avg_performance` AND `avg_availability` <= `avg_quality` THEN 'availability'
    WHEN `avg_performance` <= `avg_availability` AND `avg_performance` <= `avg_quality` THEN 'performance'
    ELSE 'quality'
  END AS `biggest_loss_factor`,
  `avg_availability` * 100 AS `availability_pct`,
  `avg_performance` * 100 AS `performance_pct`,
  `avg_quality` * 100 AS `quality_pct`
FROM ranked
WHERE `oee_rank` = 1
ORDER BY `work_center`
